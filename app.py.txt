def login_required(f):
    @wraps(f)
    def decorated_function(*args, **kwargs):
        if 'user_id' not in session:
            flash('Please log in to access this page.', 'warning')
            return redirect(url_for('login', next=request.url))
            
        # Verify user still exists and is active
        user = db.get_user_by_id(session['user_id'])
        if not user or user['is_active'] == 0:
            session.clear()
            flash('Your account has been deactivated or suspended. Please contact customer support.', 'danger')
            return redirect(url_for('login'))
        return f(*args, **kwargs)
    return decorated_function

def admin_required(f):
    @wraps(f)
    def decorated_function(*args, **kwargs):
        if 'admin_id' not in session:
            flash('Administrator authentication required.', 'warning')
            return redirect(url_for('admin_login', next=request.url))
        return f(*args, **kwargs)
    return decorated_function

@app.context_processor
def inject_user():
    """Injects current logged-in user or admin into all templates."""
    current_user = None
    current_admin = None
    if 'user_id' in session:
        current_user = db.get_user_by_id(session['user_id'])
    if 'admin_id' in session:
        current_admin = {
            'id': session.get('admin_id'),
            'username': session.get('admin_username'),
            'full_name': session.get('admin_name'),
            'role': session.get('admin_role')
        }
    return dict(current_user=current_user, current_admin=current_admin, now=datetime.utcnow())

@app.route('/register', methods=['GET', 'POST'])
def register():
    """User Registration with live webcam face enrollment."""
    if 'user_id' in session:
        return redirect(url_for('dashboard'))
        
    if request.method == 'POST':
        full_name = request.form.get('full_name', '').strip()
        email = request.form.get('email', '').strip().lower()
        phone = request.form.get('phone', '').strip()
        password = request.form.get('password', '')
        confirm_password = request.form.get('confirm_password', '')
        face_data = request.form.get('face_data', '')  # Base64 image from webcam
        
        # Validations
        if not full_name or not email or not phone or not password:
            flash('All fields are required.', 'danger')
            return render_template('register.html')
            
        if password != confirm_password:
            flash('Passwords do not match.', 'danger')
            return render_template('register.html')
            
        if len(password) < 6:
            flash('Password must be at least 6 characters long.', 'danger')
            return render_template('register.html')
            
        # Check if email is already registered
        existing_user = db.get_user_by_email(email)
        if existing_user:
            flash('An account with this email address already exists. Please log in.', 'warning')
            return redirect(url_for('login'))
            
        # Process face image
        face_encoding_json = None
        face_image_path = None
        
        if face_data:
            cv2_img, decode_err = base64_to_cv2(face_data)
            if cv2_img is not None:
                # Anti-spoofing and quality check
                passed_live, details, live_msg = check_liveness_and_quality(cv2_img)
                if not passed_live:
                    flash(f'Facial Enrollment Notice: {live_msg}', 'warning')
                    return render_template('register.html')
                    
                # Extract 128-d face embedding
                success, encoding, err_msg, _ = extract_face_encoding(cv2_img)
                if success:
                    face_encoding_json = json.dumps(encoding)
                    face_image_path = save_face_image(cv2_img, email.replace('@', '_').replace('.', '_'), "profile")
                else:
                    flash(f'Facial Feature Extraction Failed: {err_msg}', 'danger')
                    return render_template('register.html')
            else:
                flash(f'Error processing face capture: {decode_err}', 'danger')
                return render_template('register.html')
        else:
            flash('Face enrollment is required for security! Please capture your face using the webcam.', 'warning')
            return render_template('register.html')
            
        # Hash password and create user
        password_hash = generate_password_hash(password)
        user_id = db.create_user(full_name, email, phone, password_hash, face_encoding_json, face_image_path)
        
        if user_id:
            # Add a default welcome credit card for instant testing
            db.add_credit_card(
                user_id=user_id,
                cardholder_name=full_name,
                raw_card_number="4111222233334444",
                expiry_month="12",
                expiry_year="2028",
                card_limit=5000.00
            )
            
            db.log_security_event('FACE_ENROLL', user_id, 'User registered with biometric face profile', request.remote_addr)
            flash('Registration successful! Your Face ID has been enrolled and a starter credit card has been linked.', 'success')
            
            # Auto-login
            session['user_id'] = user_id
            session['user_name'] = full_name
            session['user_email'] = email
            return redirect(url_for('dashboard'))
        else:
            flash('Error creating account. Please try again.', 'danger')
            return render_template('register.html')
            
    return render_template('register.html')

@app.route('/login', methods=['GET', 'POST'])
def login():
    """User Login route."""
    if 'user_id' in session:
        return redirect(url_for('dashboard'))
        
    if request.method == 'POST':
        email = request.form.get('email', '').strip().lower()
        password = request.form.get('password', '')
        
        if not email or not password:
            flash('Please enter both email and password.', 'warning')
            return render_template('login.html')
            
        user = db.get_user_by_email(email)
        if user and check_password_hash(user['password_hash'], password):
            if user['is_active'] == 0:
                flash('Your account has been temporarily suspended due to security flags. Please contact admin.', 'danger')
                return render_template('login.html')
                
            session['user_id'] = user['id']
            session['user_name'] = user['full_name']
            session['user_email'] = user['email']
            
            db.log_security_event('LOGIN', user['id'], 'User logged in successfully', request.remote_addr)
            flash(f'Welcome back, {user["full_name"]}!', 'success')
            
            next_url = request.args.get('next')
            return redirect(next_url or url_for('dashboard'))
        else:
            flash('Invalid email or password.', 'danger')
            
    return render_template('login.html')

@app.route('/api/verify_face_transaction', methods=['POST'])
@login_required
def api_verify_face_transaction():
    """
    Core Biometric Payment Processing API:
    1. Validates inputs & card balance.
    2. Runs anti-spoofing and liveness heuristics on live webcam frame.
    3. Compares live facial encoding with registered database profile.
    4. Approves or Rejects transaction based on verification confidence.
    """
    user_id = session['user_id']
    user = db.get_user_by_id(user_id)
    
    if not user or user['is_active'] == 0:
        return jsonify({'success': False, 'message': 'Account is inactive or suspended.'}), 403
        
    data = request.get_json(silent=True) or request.form
    card_id = data.get('card_id')
    merchant_name = data.get('merchant_name', 'Online Merchant').strip()
    category = data.get('category', 'General')
    amount_str = data.get('amount')
    image_data = data.get('image_data')  # Base64 live capture
    
    # Validate fields
    if not card_id or not amount_str or not image_data:
        return jsonify({'success': False, 'message': 'Incomplete payment payload or missing camera capture.'}), 400
        
    try:
        amount = float(amount_str)
        if amount <= 0:
            return jsonify({'success': False, 'message': 'Transaction amount must be greater than zero.'}), 400
    except ValueError:
        return jsonify({'success': False, 'message': 'Invalid transaction amount.'}), 400
        
    # Check card
    card = db.get_card_by_id(card_id, user_id)
    if not card:
        return jsonify({'success': False, 'message': 'Selected credit card not found or unauthorized.'}), 404
        
    if card['available_balance'] < amount:
        return jsonify({
            'success': False,
            'message': f'Insufficient credit balance. Available: ${card["available_balance"]:.2f}, Required: ${amount:.2f}'
        }), 400
        
    # Check registered face encoding
    if not user['face_encoding']:
        return jsonify({
            'success': False,
            'message': 'No registered facial biometric profile found. Please register Face ID in Profile settings.'
        }), 400
        
    try:
        stored_encoding = json.loads(user['face_encoding'])
    except Exception:
        return jsonify({'success': False, 'message': 'Stored face profile corrupted. Please re-enroll face.'}), 500
        
    # 1. Decode live webcam frame
    cv2_img, decode_err = base64_to_cv2(image_data)
    if cv2_img is None:
        txn_id = db.generate_transaction_id()
        db.record_transaction(
            transaction_id=txn_id, user_id=user_id, card_id=card_id,
            merchant_name=merchant_name, category=category, amount=amount,
            status='FAILED', failure_reason=f'Camera capture error: {decode_err}',
            face_match_score=0.0, face_verified=0, liveness_checked=0,
            ip_address=request.remote_addr
        )
        return jsonify({'success': False, 'message': f'Image decode error: {decode_err}'}), 400
        
    # 2. Anti-spoofing and Liveness Verification
    liveness_ok, liveness_details, liveness_msg = check_liveness_and_quality(cv2_img)
    if not liveness_ok:
        txn_id = db.generate_transaction_id()
        fail_attempts = db.increment_failed_face_attempts(user_id)
        
        status = 'SUSPICIOUS' if fail_attempts >= Config.MAX_FAILED_ATTEMPTS_BEFORE_FLAG else 'FAILED'
        db.record_transaction(
            transaction_id=txn_id, user_id=user_id, card_id=card_id,
            merchant_name=merchant_name, category=category, amount=amount,
            status=status, failure_reason=f'Anti-Spoofing / Liveness Check Failed: {liveness_msg}',
            face_match_score=0.0, face_verified=0, liveness_checked=0,
            ip_address=request.remote_addr
        )
        db.log_security_event('FACE_VERIFY_FAIL', user_id, f'Liveness failure: {liveness_msg}', request.remote_addr)
        
        return jsonify({
            'success': False,
            'status': status,
            'transaction_id': txn_id,
            'message': f'Anti-Spoofing / Quality Check Failed: {liveness_msg}',
            'score': 0.0,
            'liveness': liveness_details
        }), 403
        
    # 3. Extract live facial feature vector
    success, live_encoding, feat_msg, _ = extract_face_encoding(cv2_img)
    if not success:
        txn_id = db.generate_transaction_id()
        fail_attempts = db.increment_failed_face_attempts(user_id)
        status = 'FAILED'
        db.record_transaction(
            transaction_id=txn_id, user_id=user_id, card_id=card_id,
            merchant_name=merchant_name, category=category, amount=amount,
            status=status, failure_reason=f'Facial Feature Detection Failed: {feat_msg}',
            face_match_score=0.0, face_verified=0, liveness_checked=1,
            ip_address=request.remote_addr
        )
        return jsonify({
            'success': False,
            'status': status,
            'transaction_id': txn_id,
            'message': f'Facial Feature Extraction Failed: {feat_msg}',
            'score': 0.0
        }), 403
        
    # 4. Compare live encoding against registered user encoding
    is_match, distance, confidence, reason = compare_face_encodings(stored_encoding, live_encoding)
    txn_id = db.generate_transaction_id()
    
    if is_match:
        # Deduct balance
        deduct_ok, bal_res = db.deduct_card_balance(card_id, amount)
        if not deduct_ok:
            return jsonify({'success': False, 'message': f'Balance deduction error: {bal_res}'}), 400
            
        # Reset failed attempts
        db.reset_failed_face_attempts(user_id)
        
        # Record successful transaction
        row_id = db.record_transaction(
            transaction_id=txn_id, user_id=user_id, card_id=card_id,
            merchant_name=merchant_name, category=category, amount=amount,
            status='SUCCESS', failure_reason=None,
            face_match_score=confidence, face_verified=1, liveness_checked=1,
            ip_address=request.remote_addr
        )
        db.log_security_event('FACE_VERIFY_SUCCESS', user_id, f'Transaction {txn_id} approved (${amount:.2f})', request.remote_addr)
        
        return jsonify({
            'success': True,
            'status': 'SUCCESS',
            'transaction_id': txn_id,
            'internal_id': row_id,
            'amount': f"{amount:.2f}",
            'merchant': merchant_name,
            'card_masked': card['card_number_masked'],
            'new_balance': f"{bal_res:.2f}",
            'score': confidence,
            'distance': distance,
            'message': 'Face authentication successful. Credit card transaction approved!'
        })
    else:
        # Facial Mismatch: REJECT TRANSACTION
        fail_attempts = db.increment_failed_face_attempts(user_id)
        status = 'SUSPICIOUS' if (fail_attempts >= Config.MAX_FAILED_ATTEMPTS_BEFORE_FLAG or confidence < 40.0) else 'FAILED'
        
        db.record_transaction(
            transaction_id=txn_id, user_id=user_id, card_id=card_id,
            merchant_name=merchant_name, category=category, amount=amount,
            status=status, failure_reason=reason,
            face_match_score=confidence, face_verified=0, liveness_checked=1,
            ip_address=request.remote_addr
        )
        
        db.log_security_event(
            'SUSPICIOUS_TXN' if status == 'SUSPICIOUS' else 'FACE_VERIFY_FAIL',
            user_id,
            f'Biometric mismatch on card {card["card_number_masked"]} for ${amount:.2f}. Score: {confidence}%',
            request.remote_addr
        )
        
        return jsonify({
            'success': False,
            'status': status,
            'transaction_id': txn_id,
            'amount': f"{amount:.2f}",
            'merchant': merchant_name,
            'score': confidence,
            'distance': distance,
            'message': f'Face Authentication Failed: Unauthorized individual or face does not match account owner (Match Confidence: {confidence}%).'
        }), 403

@app.route('/profile', methods=['GET', 'POST'])
@login_required
def profile():
    """User profile and Face ID management."""
    user_id = session['user_id']
    user = db.get_user_by_id(user_id)
    
    if request.method == 'POST':
        action = request.form.get('action')
        if action == 'update_info':
            full_name = request.form.get('full_name', '').strip()
            phone = request.form.get('phone', '').strip()
            if full_name and phone:
                db.update_user_profile(user_id, full_name, phone)
                session['user_name'] = full_name
                flash('Profile information updated successfully.', 'success')
            else:
                flash('Please provide both name and phone.', 'warning')
        elif action == 'update_face':
            face_data = request.form.get('face_data', '')
            if face_data:
                cv2_img, decode_err = base64_to_cv2(face_data)
                if cv2_img is not None:
                    passed_live, _, live_msg = check_liveness_and_quality(cv2_img)
                    if not passed_live:
                        flash(f'Facial Update Warning: {live_msg}', 'warning')
                        return redirect(url_for('profile'))
                        
                    success, encoding, err_msg, _ = extract_face_encoding(cv2_img)
                    if success:
                        face_encoding_json = json.dumps(encoding)
                        face_image_path = save_face_image(cv2_img, user['id'], "profile_update")
                        db.update_user_face(user_id, face_encoding_json, face_image_path)
                        db.log_security_event('FACE_ENROLL', user_id, 'Face biometric profile updated', request.remote_addr)
                        flash('Face ID successfully re-enrolled!', 'success')
                    else:
                        flash(f'Face feature extraction failed: {err_msg}', 'danger')
                else:
                    flash(f'Error processing face image: {decode_err}', 'danger')
            else:
                flash('No webcam capture received. Please capture your face.', 'warning')
                
        return redirect(url_for('profile'))
        
    return render_template('profile.html', user=user)

@app.route('/admin/users/toggle/<int:user_id>', methods=['POST'])
@admin_required
def admin_toggle_user(user_id):
    """Toggles user active / suspended status."""
    new_status = db.toggle_user_status(user_id)
    if new_status is not None:
        status_text = "Activated" if new_status == 1 else "Suspended"
        flash(f'User account {status_text} successfully.', 'info')
    else:
        flash('User not found.', 'danger')
    return redirect(url_for('admin_users'))

@app.route('/admin/transactions')
@admin_required
def admin_transactions():
    """Admin view of all transactions with filtering."""
    status_filter = request.args.get('status', 'ALL')
    search_query = request.args.get('q', '').strip()
    txns = db.get_all_transactions(status_filter=status_filter, search_query=search_query, limit=100)
    return render_template('admin_transactions.html',
                           transactions=txns,
                           current_filter=status_filter,
                           search_query=search_query)

@app.route('/admin/security-logs')
@admin_required
def admin_security_logs():
    """Admin view for audit trail and security events."""
    logs = db.get_recent_security_logs(limit=50)
    return render_template('admin_security_logs.html', logs=logs)

@app.route('/uploads/<path:filename>')
def uploaded_file(filename):
    """Serves uploaded face images."""
    return send_from_directory(Config.UPLOAD_FOLDER, filename)

@app.errorhandler(404)
def not_found_error(error):
    return render_template('404.html'), 404

@app.errorhandler(500)
def internal_error(error):
    return render_template('500.html'), 500
