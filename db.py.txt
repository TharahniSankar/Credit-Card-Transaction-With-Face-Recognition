def get_db_connection():
    """Returns a SQLite connection with row factory enabled."""
    conn = sqlite3.connect(Config.DATABASE_PATH)
    conn.row_factory = sqlite3.Row
    conn.execute("PRAGMA foreign_keys = ON")
    return conn

def init_db():
    """Initializes the database schema and creates tables if they do not exist."""
    schema_path = os.path.join(os.path.dirname(__file__), 'schema.sql')
    with open(schema_path, 'r') as f:
        schema_sql = f.read()
        
    conn = get_db_connection()
    try:
        conn.executescript(schema_sql)
        conn.commit()
    finally:
        conn.close()

def create_user(full_name, email, phone, password_hash, face_encoding_json=None, face_image_path=None):
    """Inserts a new registered user."""
    conn = get_db_connection()
    try:
        cursor = conn.cursor()
        cursor.execute("""
            INSERT INTO users (full_name, email, phone, password_hash, face_encoding, face_image_path)

def mask_card_number(raw_card_number):
    """
    Masks a 16-digit card number: keeps only last 4 digits visible.
    Example: '1234567812345678' -> '**** **** **** 5678'
    """
    clean_num = ''.join(c for c in raw_card_number if c.isdigit())
    if len(clean_num) < 4:
        return "****"
    last4 = clean_num[-4:]
    return f"**** **** **** {last4}"

def hash_card_number(raw_card_number):
    """Hashes card number with SHA-256 for duplicate detection without storing plain text."""
    clean_num = ''.join(c for c in raw_card_number if c.isdigit())
    return hashlib.sha256(clean_num.encode('utf-8')).hexdigest()

def detect_card_network(card_number):
    """Detects card brand from leading digits."""
    clean = ''.join(c for c in card_number if c.isdigit())
    if clean.startswith('4'):
        return 'Visa'
    elif clean.startswith(('51', '52', '53', '54', '55')) or (len(clean) >= 4 and 2221 <= int(clean[:4]) <= 2720):
        return 'MasterCard'
    elif clean.startswith(('34', '37')):
        return 'American Express'
    elif clean.startswith(('6011', '65')) or clean.startswith(('644', '645')):
        return 'Discover'
    elif clean.startswith(('60', '6521', '6522', '508')):
        return 'RuPay'
    return 'Credit Card'

def add_credit_card(user_id, cardholder_name, raw_card_number, expiry_month, expiry_year, card_limit=5000.00):
    """Adds a new credit card securely, storing ONLY masked digits and hash."""
    clean_num = ''.join(c for c in raw_card_number if c.isdigit())
    if len(clean_num) < 12 or len(clean_num) > 19:
        return False, "Invalid card number length (must be 12-19 digits)."
        
    masked = mask_card_number(clean_num)
    hashed = hash_card_number(clean_num)
    last4 = clean_num[-4:]
    network = detect_card_network(clean_num)
    
    conn = get_db_connection()
    try:
        # Check if user already added this exact card
        existing = conn.execute("""
            SELECT id FROM credit_cards WHERE user_id = ? AND card_number_hash = ?
        """, (user_id, hashed)).fetchone()
        
        if existing:
            return False, "This credit card is already registered to your account."
            
        # Check if user has any existing cards; if none, make this primary
        count = conn.execute("SELECT COUNT(*) as c FROM credit_cards WHERE user_id = ?", (user_id,)).fetchone()['c']
        is_primary = 1 if count == 0 else 0
        
        cursor = conn.cursor()
        cursor.execute("""
            INSERT INTO credit_cards (user_id, cardholder_name, card_number_masked, card_number_hash, last4,
                                      card_network, expiry_month, expiry_year, card_limit, available_balance, is_primary)
            VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
        """, (user_id, cardholder_name.upper().strip(), masked, hashed, last4,
              network, expiry_month.zfill(2), expiry_year, card_limit, card_limit, is_primary))
        card_id = cursor.lastrowid
        conn.commit()
        return True, card_id
    except Exception as ex:
        return False, f"Failed to add card: {str(ex)}"
    finally:
        conn.close()

def set_primary_card(user_id, card_id):
    """Marks a card as primary for the user."""
    conn = get_db_connection()
    try:
        conn.execute("UPDATE credit_cards SET is_primary = 0 WHERE user_id = ?", (user_id,))
        conn.execute("UPDATE credit_cards SET is_primary = 1 WHERE id = ? AND user_id = ?", (card_id, user_id))
        conn.commit()
        return True
    finally:
        conn.close()

def deduct_card_balance(card_id, amount):
    """Deducts approved transaction amount from available card balance."""
    conn = get_db_connection()
    try:
        card = conn.execute("SELECT available_balance FROM credit_cards WHERE id = ?", (card_id,)).fetchone()
        if not card or card['available_balance'] < amount:
            return False, "Insufficient credit balance."
            
        new_balance = round(card['available_balance'] - amount, 2)
        conn.execute("UPDATE credit_cards SET available_balance = ? WHERE id = ?", (new_balance, card_id))
        conn.commit()
        return True, new_balance
    finally:
        conn.close()

def generate_transaction_id():
    """Generates unique formatted Transaction ID (e.g. TXN-20261121-8B3E)."""
    import random
    import string
    now_str = datetime.now().strftime("%Y%m%d")
    random_hex = ''.join(random.choices(string.ascii_uppercase + string.digits, k=6))
    return f"TXN-{now_str}-{random_hex}"

def record_transaction(transaction_id, user_id, card_id, merchant_name, category, amount,
                       status, failure_reason=None, face_match_score=0.0, face_verified=0,
                       liveness_checked=0, ip_address="127.0.0.1"):
    """Inserts a transaction entry."""
    conn = get_db_connection()
    try:
        cursor = conn.cursor()
        cursor.execute("""
            INSERT INTO transactions (transaction_id, user_id, card_id, merchant_name, category, amount,
                                      status, failure_reason, face_match_score, face_verified, liveness_checked, ip_address)
            VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
        """, (transaction_id, user_id, card_id, merchant_name, category, float(amount),
              status, failure_reason, float(face_match_score), int(face_verified), int(liveness_checked), ip_address))
        txn_id = cursor.lastrowid
        conn.commit()
        return txn_id
    finally:
        conn.close()

def get_transaction_by_id(txn_id):
    """Retrieves transaction details along with card and user info."""

            SELECT t.*,

                   u.full_name as user_name, u.email as user_email

            LEFT JOIN users u ON t.user_id = u.id

def get_user_transactions(user_id, status_filter=None, search_query=None, limit=50):
    """Fetches transaction history for a specific user with filtering."""
    conn = get_db_connection()
    try:
        query = """
            SELECT t.*, c.card_number_masked, c.card_network
            FROM transactions t
            LEFT JOIN credit_cards c ON t.card_id = c.id
            WHERE t.user_id = ?
        """
        params = [user_id]
        
        if status_filter and status_filter.upper() != 'ALL':
            query += " AND t.status = ?"
            params.append(status_filter.upper())
            
        if search_query:
            query += " AND (t.merchant_name LIKE ? OR t.transaction_id LIKE ?)"
            params.append(f"%{search_query}%")
            params.append(f"%{search_query}%")
            
        query += " ORDER BY t.created_at DESC LIMIT ?"
        params.append(limit)
        
        return conn.execute(query, params).fetchall()
    finally:
        conn.close()

def get_dashboard_metrics():
    """Calculates overall metrics for Admin Dashboard."""
    conn = get_db_connection()
    try:
        total_users = conn.execute("SELECT COUNT(*) as c FROM users").fetchone()['c']
        total_cards = conn.execute("SELECT COUNT(*) as c FROM credit_cards").fetchone()['c']
        
        txn_stats = conn.execute("""
            SELECT 
                COUNT(*) as total_txns,
                SUM(CASE WHEN status = 'SUCCESS' THEN 1 ELSE 0 END) as success_txns,
                SUM(CASE WHEN status = 'FAILED' THEN 1 ELSE 0 END) as failed_txns,
                SUM(CASE WHEN status = 'SUSPICIOUS' THEN 1 ELSE 0 END) as suspicious_txns,
                SUM(CASE WHEN status = 'SUCCESS' THEN amount ELSE 0 END) as total_volume
            FROM transactions
        """).fetchone()
        
        total_txns = txn_stats['total_txns'] or 0
        success_txns = txn_stats['success_txns'] or 0
        failed_txns = txn_stats['failed_txns'] or 0
        suspicious_txns = txn_stats['suspicious_txns'] or 0
        total_volume = txn_stats['total_volume'] or 0.0
        
        success_rate = round((success_txns / total_txns * 100), 1) if total_txns > 0 else 100.0
        
        # Recent 7 days volume query for chart
        daily_trends = conn.execute("""
            SELECT DATE(created_at) as txn_date, 
                   COUNT(*) as total_count,
                   SUM(CASE WHEN status = 'SUCCESS' THEN amount ELSE 0 END) as daily_volume
            FROM transactions
            GROUP BY DATE(created_at)
            ORDER BY txn_date DESC
            LIMIT 7
        """).fetchall()
        
        return {
            "total_users": total_users,
            "total_cards": total_cards,
            "total_txns": total_txns,
            "success_txns": success_txns,
            "failed_txns": failed_txns,
            "suspicious_txns": suspicious_txns,
            "total_volume": round(total_volume, 2),
            "success_rate": success_rate,
            "daily_trends": [dict(r) for r in daily_trends]
        }
    finally:
        conn.close()
